\begin{abstract}
Let $P(n)$ count permutations $p$ of $\{1,\dots,n\}$
such that $i+p(i)$ is a perfect cube for every $i$.
We establish $P(n)>0$ for all $n\ge173$ by
a short reflection argument and $109$ explicit
finite integer identities. A general boundary-cancellation
lemma converts each finite instance into
an equality of two multisets of endpoints.
No matching algorithm, large permutation array,
or sequence of dependent augmenting paths is
required to \emph{check} the certificate.
The proof is elementary but retains a finite
$109$-row table; it is not a uniform case-free
construction for the complete base interval.
\end{abstract}

\section{The boundary-cancellation principle}

\begin{lemma}[Integer endpoint balance]
\label{lem:balance}
Let $n\ge1$, choose a partition
$0=b_0<b_1<\cdots<b_t=n$
and positive perfect cubes $c_1,\ldots,c_t$.
Put $L_j=c_j-b_j$ and
$R_j=c_j-b_{j-1}$.
Assume $1\le L_j<R_j\le n+1$.
The piecewise reflection
\[
p(i)=c_j-i,\qquad b_{j-1}<i\le b_j
\]
is a cube-sum permutation of $[1,n]$
if and only if
\begin{equation}\label{eq:balance}
\boxed{\{1,R_1,\ldots,R_t\}_{\rm multi}
=\{n+1,L_1,\ldots,L_t\}_{\rm multi}.}
\end{equation}
\end{lemma}
\begin{proof}
On input interval $(b_{j-1},b_j]$, the map is
a bijection onto the integer interval
$[L_j,R_j)$, with sum $i+p(i)=c_j$.
Write $f(x)$ for the number of output
intervals covering $x$. The jump
$f(x)-f(x-1)$ equals the multiplicity
of $x$ among left endpoints minus
its multiplicity among right endpoints.
Identity \eqref{eq:balance} says that
this jump is $1$ at $x=1$,
$-1$ at $x=n+1$, and zero otherwise.
Thus $f(x)=1$ for every $1\le x\le n$.
Every output occurs exactly once,
and the cube-sum condition holds.
The converse is immediate from the
same endpoint-counting identity.
\end{proof}

\section{The finite table}

In Appendix~\ref{app:table}, a row labelled $n$
has tokens $b:k$. They mean $b_j=b$,
$c_j=k^3$, and the preceding boundary
is $b_{j-1}$ (the first boundary is $0$).
The last token ends at $b_t=n$.

\begin{lemma}[The base interval]
\label{lem:base}
For every $n$ satisfying
\[
173\le n\le255
\quad\text{or}\quad
339\le n\le364,
\]
there exists a cube-sum permutation
of $[1,n]$.
\end{lemma}
\begin{proof}
The appendix records $109$ explicit rows
using cube roots $k\in\{2,\ldots,8\}$.
Every row has strictly increasing boundaries,
all $1\le L_j<R_j\le n+1$, and the equality
of multisets in \eqref{eq:balance}.
These are direct integer identities:
the entire table consists of $1123$
intervals. Lemma~\ref{lem:balance}
therefore proves the claim without
any matching search.

For example, the row $n=340$ is
\[
7:2,\quad19:3,\quad44:4,\quad171:6,\quad340:8.
\]
The two endpoint lists, after sorting,
both equal
\[
(1,8,20,45,172,341).
\]
For the exceptional $n=236$, the
row with $15$ pieces gives the common
sorted endpoint list
\[
\begin{gathered}
1,8,10,19,44,46,71,82,\\
107,135,137,171,173,198,209,237.
\end{gathered}
\]
The same elementary equality is checked
for all the remaining rows.
\end{proof}

\section{Reflection and the infinite tail}

\begin{lemma}[Suffix reflection]
\label{lem:suffix}
If $0\le m<n$ and $m+n+1=c=k^3$,
then every cube-sum permutation on
$[1,m]$ extends to $[1,n]$ by
\[
p(i)=
\begin{cases}
p_m(i),&1\le i\le m,\\
c-i,&m<i\le n.
\end{cases}
\]
For $m=0$, the empty permutation is used.
\end{lemma}
\begin{proof}
The new interval $[m+1,n]$
is invariant under $i\mapsto c-i$,
and all its index-image sums equal $c$.
\end{proof}

\begin{theorem}[All lengths from $173$ onward]
\label{thm:main}
For every integer $n\ge173$,
there is a permutation $p$ of $[1,n]$
such that every $i+p(i)$ is
a perfect cube.
\end{theorem}
\begin{proof}
Lemma~\ref{lem:base} covers
$173\le n\le255$ and $339\le n\le364$.

For $256\le n\le338$, take
$m=511-n\in[173,255]$.
Then $m<n$ and $m+n+1=512=8^3$.
Lemma~\ref{lem:suffix} covers
these $83$ remaining finite cases.

For $365\le n\le555$, take
$c=729=9^3$.
The predecessor $m=728-n$
satisfies $173\le m<n$, so
the suffix reflection gives the result
by strong induction.

For $n\ge556$, let $k$ be the least
integer with $k^3\ge n+174$,
and put $m=k^3-n-1$.
By construction $m\ge173$.
Since $k\ge10$, the minimality of $k$
gives $n\ge(k-1)^3-173$. Hence
\[
\begin{aligned}
2n-k^3
&\ge2((k-1)^3-173)-k^3\\
&=k^3-6k^2+6k-348\\
&\ge112>0.
\end{aligned}
\]
The last polynomial takes value $112$
at $k=10$ and has strictly positive
successive differences $3k^2-9k+1$
for $k\ge10$. Thus $k^3\le2n$,
whence $m<n$.
The suffix-reflection lemma and
strong induction finish the proof.
\end{proof}

\paragraph{Scope.}
The mathematical validity of this proof
reduces to the displayed elementary lemma,
$109$ explicit integer-multiset equalities,
and the two short induction cases.
The finite table is indispensable to the
proof in this form: it does not assert
a hidden closed formula or a
case-free argument for $173\le n\le364$.
